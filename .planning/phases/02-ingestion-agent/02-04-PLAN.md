    **Helper function — get_recording_timestamp (per D-21, Pitfall 7):**
    ```python
    def get_recording_timestamp(audio_path: Path) -> datetime:
        """Extract recording timestamp from file mtime. Fallback to now.

        Per D-21: read from filesystem metadata.
        Per Pitfall 7: Linux has no st_birthtime — use st_mtime.
        """
        try:
            stat = audio_path.stat()
            mtime = stat.st_mtime
            return datetime.fromtimestamp(mtime, tz=timezone.utc)
        except (OSError, ValueError):
            return datetime.now(timezone.utc)
    ```

    **Helper function — track modified IDs (for compression pass):**

    The agent tools return dicts with "node_id" or "edge_id" on create/update. To track these, wrap the tools dict to capture IDs:

    ```python
    def tracking_wrapper(tools_dict: dict) -> tuple[dict, list, list]:
        """Wrap tool callables to track which node/edge IDs were created or updated."""
        modified_node_ids = []
        modified_edge_ids = []

        wrapped = {}
        for name, fn in tools_dict.items():
            if name in ("create_node", "update_node"):
                def make_wrapper(original, id_list, key="node_id"):
                    def wrapper(**kwargs):
                        result = original(**kwargs)
                        if isinstance(result, dict) and key in result:
                            id_list.append(result[key])
                        return result
                    return wrapper
                wrapped[name] = make_wrapper(fn, modified_node_ids, "node_id")
            elif name in ("create_edge", "update_edge"):
                def make_wrapper(original, id_list, key="edge_id"):
                    def wrapper(**kwargs):
                        result = original(**kwargs)
                        if isinstance(result, dict) and key in result:
                            id_list.append(result[key])
                        return result
                    return wrapper
                wrapped[name] = make_wrapper(fn, modified_edge_ids, "edge_id")
            else:
                wrapped[name] = fn

        return wrapped, modified_node_ids, modified_edge_ids
    ```
